ШАГ 1: Закройте Cursor полностью (если открыт)

ШАГ 2: Откройте PowerShell и выполните:

$cursorPath = "$env:APPDATA\Cursor\User\globalStorage"
$backup = Join-Path $cursorPath "state.vscdb.backup"
$main = Join-Path $cursorPath "state.vscdb"

# Проверяем размеры файлов
if (Test-Path $backup) {
    $backupSize = (Get-Item $backup).Length
    $mainSize = (Get-Item $main).Length
    
    Write-Host "Резервная копия: $([math]::Round($backupSize/1KB, 2)) KB"
    Write-Host "Основной файл: $([math]::Round($mainSize/1KB, 2)) KB"
    
    if ($backupSize -gt $mainSize) {
        Write-Host ""
        Write-Host "✓ Резервная копия БОЛЬШЕ! Можно восстановить." -ForegroundColor Green
        Write-Host ""
        $answer = Read-Host "Восстановить? (y/n)"
        if ($answer -eq 'y') {
            # Сохраняем текущий файл
            Copy-Item $main "$main.old_$(Get-Date -Format 'yyyyMMdd_HHmmss')"
            # Восстанавливаем
            Copy-Item $backup $main -Force
            Write-Host "✓ Готово! Откройте Cursor." -ForegroundColor Green
        }
    } else {
        Write-Host "Резервная копия не больше основного файла." -ForegroundColor Yellow
    }
}
